# Data Schema — Input / Output Contracts

## Raw Transaction (Layer 1 Output → Layer 2 Input)

```typescript
interface RawTransaction {
  extraction_id: string;        // uuid
  pipeline_run_id: string;      // uuid, groups all records from one extraction
  source_bank: string;          // "BNU" | "HSBC" | "BPI" etc.
  extracted_at: string;         // ISO 8601 timestamp
  raw_date: string;             // as-scraped, may be "15/03/2026" or "Mar 15"
  raw_description: string;      // as-scraped, unmodified
  raw_debit: string | null;     // as-scraped string, e.g. "1,250.00"
  raw_credit: string | null;
  raw_balance: string | null;
  raw_reference: string | null;
  source_page: number;          // page number in statement if PDF
  source_file: string | null;   // filename if from PDF
}
```

## Classified Transaction (Layer 2 Output → Data Warehouse)

```typescript
interface ClassifiedTransaction {
  // Identity
  transaction_id: string;       // uuid, generated by Layer 2
  pipeline_run_id: string;      // links back to raw extraction
  source_bank: string;

  // Normalized fields
  transaction_date: Date;
  description: string;          // cleaned description
  debit_amount: number | null;
  credit_amount: number | null;
  balance: number | null;
  currency: string;             // "MOP" | "HKD" | "USD" etc.
  reference_number: string | null;

  // AI Classification
  category: string;             // e.g. "Trade Payment"
  sub_category: string;         // e.g. "Raw Materials"
  gl_code: string;              // e.g. "5100"
  cost_center: string;          // e.g. "OPERATIONS"
  classification_confidence: number;  // 0.0 - 1.0
  classification_model: string; // "claude-sonnet-4-6" | "gpt-4o" etc.
  classified_at: string;        // ISO 8601

  // Validation
  validation_status: "PASSED" | "FLAGGED" | "REJECTED";
  validation_flags: string[];   // e.g. ["DUPLICATE_SUSPECTED", "LOW_CONFIDENCE"]
  requires_human_review: boolean;
  reviewed_by: string | null;
  reviewed_at: string | null;

  // Audit
  created_at: string;
  updated_at: string;
}
```

## Excel Report Row (Layer 3 — Client Facing)

```typescript
interface ExcelReportRow {
  Date: string;                 // "15/03/2026" — client's preferred format
  Description: string;
  Debit: number | null;
  Credit: number | null;
  Balance: number | null;
  Currency: string;
  Category: string;
  "GL Code": string;
  "Cost Center": string;
  Reference: string;
  Status: string;               // "OK" | "Needs Review"
}
```

## Pipeline Run Record (Audit Log)

```typescript
interface PipelineRun {
  run_id: string;               // uuid
  bank_id: string;
  triggered_by: string;         // client user ID
  triggered_at: string;
  layer1_status: "PENDING" | "RUNNING" | "COMPLETE" | "FAILED";
  layer1_records_extracted: number;
  layer2_status: "PENDING" | "RUNNING" | "COMPLETE" | "FAILED";
  layer2_records_classified: number;
  layer2_records_flagged: number;
  layer3_status: "PENDING" | "RUNNING" | "COMPLETE" | "FAILED";
  layer3_excel_path: string | null;
  completed_at: string | null;
  errors: string[];
}
```

## LLM Classification Request / Response

```typescript
// What we send to LLM (NO credentials, NO PII beyond transaction data)
interface ClassificationRequest {
  transaction_id: string;
  description: string;
  amount: number;
  direction: "DEBIT" | "CREDIT";
  bank: string;
  date: string;
}

// What LLM returns (enforced via system prompt)
interface ClassificationResponse {
  category: string;
  sub_category: string;
  gl_code: string;
  cost_center: string;
  confidence: number;
  reasoning: string;  // brief explanation, for audit trail
  flag: string | null;
}
```
